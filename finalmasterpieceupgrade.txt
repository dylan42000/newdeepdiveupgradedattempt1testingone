## PART 1 — ACCOUNT NUMBER PARSE & MERGE ENGINE (The Root Problem)

### 1.1 Why Account Numbers Are Truncated Today

The live parser (`creditParser.ts`) extracts what is **visible in the PDF text layer**. Bureaus intentionally mask account numbers:

- Equifax: `XXXX-XXXX-XXXX-1234` or `****1234`
- Experian: `123456XXXXXXXX` or partial 4-digit suffix only
- TransUnion: `Account #: 1234` (last 4 only in column export)

The current dedup key is:
```
bureau|creditor|accountNumber|dateOpened
```

When `accountNumber` is `****1234` on EQ and `1234` on TU for the same account, the dedup key **never matches** → duplicate `NegativeItem` entries for the same tradeline, one per bureau fragment.

### 1.2 Account Number Reconstruction Strategy

**File to create:** `src/services/creditReportParser/accountNumberReconstructor.ts`

#### Step 1 — Normalization

Strip all masking characters and normalize to a canonical suffix token:

```typescript
// accountNumberReconstructor.ts

export interface ReconstructedAccountNumber {
  raw: string;                    // original parsed string
  normalized: string;             // digits only, no mask chars
  suffix: string;                 // last 4 digits (canonical join key)
  prefix: string | null;          // leading digits if exposed
  maskPattern: MaskPattern;       // detected mask type
  confidence: number;             // 0-1 reconstruction confidence
  fullEstimate: string | null;    // best-effort full number if inferrable
}

export type MaskPattern =
  | 'suffix_4'          // ****1234  |  XXXX1234  |  1234 (bare)
  | 'suffix_6'          // xxxxxx123456
  | 'prefix_masked'     // 123456XXXXXXXX
  | 'middle_masked'     // 1234****5678
  | 'full_visible'      // no masking detected
  | 'unknown';

const MASK_CHARS = /[\*xX\-\_\s]/g;

export function normalizeAccountNumber(raw: string): ReconstructedAccountNumber {
  if (!raw || raw.trim().length === 0) {
    return makeEmpty(raw);
  }

  const cleaned = raw.trim();
  const digitsOnly = cleaned.replace(MASK_CHARS, '');
  const suffix = digitsOnly.slice(-4);
  const prefix = digitsOnly.length > 4 ? digitsOnly.slice(0, digitsOnly.length - 4) : null;
  
  const pattern = detectMaskPattern(cleaned, digitsOnly);
  const confidence = calcReconstructionConfidence(cleaned, digitsOnly, pattern);
  
  // Full estimate: if we have ≥8 visible digits we probably have the real number
  const fullEstimate = digitsOnly.length >= 8 ? digitsOnly : null;

  return {
    raw: cleaned,
    normalized: digitsOnly,
    suffix,
    prefix,
    maskPattern: pattern,
    confidence,
    fullEstimate,
  };
}

function detectMaskPattern(raw: string, digits: string): MaskPattern {
  if (/^[\d]{8,}$/.test(raw)) return 'full_visible';
  if (/^[\*xX]{4,}\d{4}$/.test(raw.replace(/[\-\s]/g, ''))) return 'suffix_4';
  if (/^[\*xX]{4,}\d{6}$/.test(raw.replace(/[\-\s]/g, ''))) return 'suffix_6';
  if (/^\d{4,}[\*xX]{4,}$/.test(raw.replace(/[\-\s]/g, ''))) return 'prefix_masked';
  if (/^\d{4}[\*xX]+\d{4}$/.test(raw.replace(/[\-\s]/g, ''))) return 'middle_masked';
  if (digits.length >= 4) return 'suffix_4'; // fallback assume suffix
  return 'unknown';
}

function calcReconstructionConfidence(raw: string, digits: string, pattern: MaskPattern): number {
  if (pattern === 'full_visible') return 1.0;
  if (digits.length === 0) return 0.0;
  if (digits.length >= 8) return 0.85;
  if (digits.length >= 6) return 0.70;
  if (digits.length === 4) return 0.55; // suffix-only, usable for merge but not unique
  return 0.30;
}
```

#### Step 3 — Weighted Merge Confidence Scoring

**File to create:** `src/services/accountMergeEngine/mergeSimilarityEngine.ts`

Replace the binary dedup key with a **weighted similarity score**:

```typescript
export interface MergeCandidate {
  itemA: NegativeItem;
  itemB: NegativeItem;
  score: number;              // 0-100
  factors: MergeFactor[];
  decision: MergeDecision;
  mergedAccountNumber: string | null;  // best reconstruction
}

export type MergeDecision = 
  | 'AUTO_MERGE'      // score ≥ 85
  | 'SUGGEST_MERGE'   // score 65–84
  | 'MANUAL_REVIEW'   // score 45–64
  | 'NO_MERGE';       // score < 45

interface MergeFactor {
  name: string;
  weight: number;
  score: number;
  contribution: number;
}

const WEIGHTS = {
  accountSuffix:      30,   // same last-4 → high signal
  creditorName:       25,   // normalized match
  accountType:        15,   // both COLLECTION or both CHARGE_OFF
  dateOpened:         15,   // within 60 days
  balance:            10,   // same bucket
  accountPrefix:       5,   // bonus if prefix also matches
} as const;

export function scoreMergeCandidates(a: NegativeItem, b: NegativeItem): MergeCandidate {
  const factors: MergeFactor[] = [];
  let totalScore = 0;

  // Suffix match (most important)
  const suffixA = extractSuffix(a.accountNumber);
  const suffixB = extractSuffix(b.accountNumber);
  const suffixScore = suffixA && suffixB && suffixA === suffixB ? 100 : 
                      suffixA && suffixB ? 0 : 50; // 50 if one is missing
  const suffixContrib = (suffixScore / 100) * WEIGHTS.accountSuffix;
  factors.push({ name: 'accountSuffix', weight: WEIGHTS.accountSuffix, score: suffixScore, contribution: suffixContrib });
  totalScore += suffixContrib;

  // Creditor name similarity (Jaro-Winkler or token overlap)
  const credScore = creditorSimilarity(a.creditorName, b.creditorName);
  const credContrib = credScore * WEIGHTS.creditorName;
  factors.push({ name: 'creditorName', weight: WEIGHTS.creditorName, score: credScore * 100, contribution: credContrib });
  totalScore += credContrib;

  // Account type match
  const typeMatch = normalizeAccountType(a.accountType) === normalizeAccountType(b.accountType);
  const typeContrib = typeMatch ? WEIGHTS.accountType : 0;
  factors.push({ name: 'accountType', weight: WEIGHTS.accountType, score: typeMatch ? 100 : 0, contribution: typeContrib });
  totalScore += typeContrib;

  // Date opened within 60 days
  const dateScore = dateProximityScore(a.dateOpened, b.dateOpened, 60);
  const dateContrib = (dateScore / 100) * WEIGHTS.dateOpened;
  factors.push({ name: 'dateOpened', weight: WEIGHTS.dateOpened, score: dateScore, contribution: dateContrib });
  totalScore += dateContrib;

  // Balance bucket
  const balMatch = bucketBalance(a.balance) === bucketBalance(b.balance);
  const balContrib = balMatch ? WEIGHTS.balance : 0;
  factors.push({ name: 'balance', weight: WEIGHTS.balance, score: balMatch ? 100 : 0, contribution: balContrib });
  totalScore += balContrib;

  // Prefix bonus
  const prefA = extractPrefix(a.accountNumber);
  const prefB = extractPrefix(b.accountNumber);
  if (prefA && prefB && prefA === prefB) {
    totalScore += WEIGHTS.accountPrefix;
    factors.push({ name: 'accountPrefix', weight: WEIGHTS.accountPrefix, score: 100, contribution: WEIGHTS.accountPrefix });
  }

  const finalScore = Math.min(100, Math.round(totalScore));
  const decision: MergeDecision = 
    finalScore >= 85 ? 'AUTO_MERGE' :
    finalScore >= 65 ? 'SUGGEST_MERGE' :
    finalScore >= 45 ? 'MANUAL_REVIEW' : 'NO_MERGE';

  // Best account number reconstruction
  const mergedAccountNumber = reconstructBestAccountNumber(a.accountNumber, b.accountNumber);

  return { itemA: a, itemB: b, score: finalScore, factors, decision, mergedAccountNumber };
}

// Cross-bureau reconstruction: take the version with MORE visible digits
export function reconstructBestAccountNumber(acctA: string, acctB: string): string | null {
  const reconA = normalizeAccountNumber(acctA);
  const reconB = normalizeAccountNumber(acctB);
  
  if (reconA.confidence >= reconB.confidence) {
    return reconA.fullEstimate ?? reconA.normalized ?? acctA;
  }
  return reconB.fullEstimate ?? reconB.normalized ?? acctB;
}
```

#### Step 4 — Auto-Merge Orchestrator

**File to create:** `src/services/accountMergeEngine/autoMergeOrchestrator.ts`

```typescript
export interface AutoMergeResult {
  mergedGroups: MergedAccountGroup[];
  suggestedMerges: MergeCandidate[];
  manualReviewItems: MergeCandidate[];
  stats: {
    totalItems: number;
    autoMerged: number;
    suggestedCount: number;
    manualCount: number;
    uniqueAccountsAfterMerge: number;
  };
}

export interface MergedAccountGroup {
  primary: NegativeItem;          // Highest-confidence version (most data)
  duplicates: NegativeItem[];     // Items merged into primary
  allBureaus: CreditBureau[];     // Union of all bureaus
  bestAccountNumber: string;      // Reconstructed full/best account number
  mergeConfidence: number;
  mergeFactors: MergeFactor[];
}

export function runAutoMerge(items: NegativeItem[]): AutoMergeResult {
  const mergedGroups: MergedAccountGroup[] = [];
  const suggestedMerges: MergeCandidate[] = [];
  const manualReviewItems: MergeCandidate[] = [];
  const usedIds = new Set<string>();

  // Sort by priority score descending (primary = highest quality)
  const sorted = [...items].sort((a, b) => (b.priorityScore ?? 0) - (a.priorityScore ?? 0));

  for (let i = 0; i < sorted.length; i++) {
    if (usedIds.has(sorted[i].id)) continue;

    const group: MergedAccountGroup = {
      primary: sorted[i],
      duplicates: [],
      allBureaus: [...(sorted[i].creditBureau ?? [])],
      bestAccountNumber: sorted[i].accountNumber,
      mergeConfidence: 1.0,
      mergeFactors: [],
    };

    for (let j = i + 1; j < sorted.length; j++) {
      if (usedIds.has(sorted[j].id)) continue;

      const candidate = scoreMergeCandidates(sorted[i], sorted[j]);

      if (candidate.decision === 'AUTO_MERGE') {
        // Merge: union bureaus, pick best account number
        group.duplicates.push(sorted[j]);
        group.allBureaus = [...new Set([...group.allBureaus, ...(sorted[j].creditBureau ?? [])])];
        group.bestAccountNumber = candidate.mergedAccountNumber ?? group.bestAccountNumber;
        group.mergeConfidence = Math.min(group.mergeConfidence, candidate.score / 100);
        group.mergeFactors = candidate.factors;
        usedIds.add(sorted[j].id);
      } else if (candidate.decision === 'SUGGEST_MERGE') {
        suggestedMerges.push(candidate);
      } else if (candidate.decision === 'MANUAL_REVIEW') {
        manualReviewItems.push(candidate);
      }
    }

    // Build final merged NegativeItem
    group.primary = {
      ...group.primary,
      accountNumber: group.bestAccountNumber,
      creditBureau: group.allBureaus,
      // Preserve the worst-case financial data for strongest dispute posture
      balance: pickHighestBalance([group.primary, ...group.duplicates]),
      dateOfFirstDelinquency: pickEarliestDOFD([group.primary, ...group.duplicates]),
    };

    usedIds.add(sorted[i].id);
    mergedGroups.push(group);
  }

  return {
    mergedGroups,
    suggestedMerges,
    manualReviewItems,
    stats: {
      totalItems: items.length,
      autoMerged: mergedGroups.reduce((n, g) => n + g.duplicates.length, 0),
      suggestedCount: suggestedMerges.length,
      manualCount: manualReviewItems.length,
      uniqueAccountsAfterMerge: mergedGroups.length,
    },
  };
}
```

```typescript
// Track used citations per profile/bureau/item
export function getNextSafeCitation(
  bureau: string,
  pass: number,
  usedCitations: string[],   // from dispute history
  dna: LetterDNA12,
): CitationEntry {
  const profile = BUREAU_CITATION_PROFILES[bureau];
  if (!profile) return FALLBACK_CITATION;

  const eligible = profile.preferredCitations
    .filter(c => c.passCompatibility.includes(pass))
    .filter(c => c.frivolousRisk === 'safe' || pass >= 3)
    .filter(c => {
      const timesUsed = usedCitations.filter(u => u === c.citation).length;
      return timesUsed < profile.avoidAfterRepeat;
    });

  // Pick by DNA seed for determinism but cycle through unused ones first
  if (eligible.length === 0) {
    // All used → pick lowest-risk never-flagged fallback
    return profile.preferredCitations.find(c => c.citation === profile.safeLeadCitation) 
           ?? FALLBACK_CITATION;
  }

  return eligible[dna.persona.charCodeAt(0) % eligible.length];
}
```

### 2.4 Fact Injection Pipeline

**File to create:** `src/services/letterFactInjector.ts`

The single biggest driver of dispute success is **specificity**. Letters that reference actual dollar amounts, actual dates, and specific field names are 3× less likely to be flagged as frivolous:

```typescript
export interface LetterFactBlock {
  creditorName: string;
  accountNumber: string;         // best reconstruction
  accountSuffix: string;         // always available
  balance: string;               // formatted: "$1,234.00"
  dofd: string | null;           // "January 2022" (month only for privacy)
  accountType: string;           // human-readable
  reportingStatus: string;       // "Charge-Off", "Collection", etc.
  bureauSpecificFieldRef: string; // EQ: "Account Condition", EX: "Account Status", TU: "Payment Status"
  openDate: string | null;
  closedDate: string | null;
  daysSinceDofd: number | null;

}

export function buildFactBlock(item: NegativeItem, bureau: string): LetterFactBlock {

  const daysSinceDofd = dofd ? Math.floor((Date.now() - dofd.getTime()) / 86400000) : null;
  const reportingDuration = dofd ? formatDuration(dofd) : null;

  return {
    creditorName: item.creditorName,
    accountNumber: item.accountNumber,
    accountSuffix: extractSuffix(item.accountNumber) ?? 'on file',
    balance: item.balance ? `$${Number(item.balance).toLocaleString('en-US', {minimumFractionDigits: 2})}` : 'an amount I dispute',
    dofd: dofd ? dofd.toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : null,
    accountType: humanizeAccountType(item.accountType),
    reportingStatus: item.accountStatus ?? item.accountType,
    bureauSpecificFieldRef: BUREAU_FIELD_NAMES[bureau]?.[item.accountType] ?? 'account status',
```

```typescript
async function evaluateAutoApproveEligibility(
  letter: GeneratedLetterV2,
  planItem: DispatchPlanItem,
  settings: AutoPilotSettingsV3,
  factBlock: LetterFactBlock,
): Promise<{ eligible: boolean; reason: string }> {
  
  // Pass ceiling check
  if (planItem.pass > settings.letterAutoApproveMaxPass) {
    return { eligible: false, reason: `pass_${planItem.pass}_requires_manual_review` };
  }
  
  // Uniqueness score check
  const uniquenessResult = await evaluateDisputeUniqueness(letter.content, planItem.item.id, planItem.bureau);
  if (uniquenessResult.score < settings.letterAutoApprovePassThreshold) {
    return { eligible: false, reason: `uniqueness_score_${uniquenessResult.score}_below_threshold` };
  }
  
  // Factual anchor check
  if (settings.requireFactualAnchorValidation) {
    const hasAnchor = letter.content.toLowerCase().includes(factBlock.accountSuffix);
    if (!hasAnchor) {
      return { eligible: false, reason: 'missing_account_suffix_anchor' };
    }
  }
  
  // No unfilled tokens
  const unfilledTokens = scanForUnfilledTokens(letter.content);
  if (unfilledTokens.length > 0) {
    return { eligible: false, reason: `unfilled_tokens:${unfilledTokens.join(',')}` };
  }
  
  // Pass 1 is always auto-approvable if above checks pass
  // Pass 2-4 require all checks AND no recent frivolous flag on this bureau
  if (planItem.pass >= 2) {
    const recentFrivolous = await DisputeHistoryService.getRecentFrivolousFlags(
      planItem.item.id, planItem.bureau, 90
    );
    if (recentFrivolous > 0) {
      return { eligible: false, reason: 'recent_frivolous_flag_on_bureau' };
    }
  }
  
  return { eligible: true, reason: 'all_gates_passed' };
}
```

```typescript
function getAdaptiveDuplicateWindowDays(
  bureau: string,
  pass: number,
  settings: AutoPilotSettingsV3,

  if (!settings.adaptiveDuplicateWindow) return 30; // legacy flat

  // Bureau expected response latency + buffer

    Equifax: 28,
    Experian: 25,
    TransUnion: 30,
  };
  
  const bureauDays = BUREAU_EXPECTED_DAYS[bureau] ?? 30;
  
  // Window = max(25, bureauDays + 5) 
  // This prevents re-dispatch before bureau could even respond
  const base = Math.max(25, bureauDays + 5);

  if (pass >= 5) return Math.min(base, 20);
  
  return base;
}
```

```typescript
export function getAdaptiveFrivolousHoldDays(
  bureau: string,
  pass: number,
  triggerReason: FrivolousReason,
  settings: AutoPilotSettingsV3,
): number {
  if (!settings.adaptiveFrivolousHold) return 14; // legacy flat

  // Base hold by bureau frivolous risk profile
  const bureauBase: Record<string, number> = {
    Experian: 21,   // High risk → longer cooldown
    Equifax: 14,    // Medium
    TransUnion: 14, // Medium
  };

  let hold = bureauBase[bureau] ?? 14;

  // Reduce hold if pass ≥ 4 (we're in escalation, don't want long delays)
  if (pass >= 4) hold = Math.min(hold, 10);
  if (pass >= 5) hold = Math.min(hold, 7);

  // Increase hold if trigger is same-basis repeat (actual frivolous risk)
  if (triggerReason === 'same_legal_basis_repeat') hold += 7;

  return hold;
}
```

```typescript
export interface FrivolousRiskAssessment {
  riskScore: number;              // 0-100
  riskLevel: 'none' | 'low' | 'medium' | 'high' | 'critical';
  triggerFactors: FrivolousRiskFactor[];
  recommendation: 'proceed' | 'proceed_with_rotation' | 'hold' | 'escalate_strategy';

  suggestedStrategy?: PassStrategy;

interface FrivolousRiskFactor {
  factor: string;
  weight: number;
  triggered: boolean;
  contribution: number;
}

const RISK_FACTORS: Array<{ name: string; weight: number; test: (ctx: FrivolousRiskContext) => boolean }> = [
  { name: 'same_lead_citation_repeat', weight: 35, test: ctx => ctx.sameLeadCitationCount >= 2 },
  { name: 'same_legal_angle_repeat',   weight: 25, test: ctx => ctx.sameLegalAngleCount >= 2 },
  { name: 'recent_activity_under_30d', weight: 20, test: ctx => ctx.daysSinceLastDispute < 30 },
  { name: 'no_new_evidence_since_last',weight: 15, test: ctx => !ctx.hasNewEvidenceSinceLast },
  { name: 'experian_high_risk_bureau', weight: 10, test: ctx => ctx.bureau === 'Experian' },

  { name: 'generic_dispute_language',  weight: 20, test: ctx => ctx.hasGenericOpeningDetected },
];

  const factors: FrivolousRiskFactor[] = RISK_FACTORS.map(f => ({
    factor: f.name,
    weight: f.weight,
    triggered: f.test(ctx),
    contribution: f.test(ctx) ? f.weight : 0,
  }));

  const riskScore = Math.min(100, factors.reduce((sum, f) => sum + f.contribution, 0));
  
  const riskLevel = riskScore >= 70 ? 'critical' :
                    riskScore >= 50 ? 'high' :
                    riskScore >= 30 ? 'medium' :
                    riskScore >= 10 ? 'low' : 'none';

  const recommendation = riskScore >= 70 ? 'hold' :
                         riskScore >= 40 ? 'proceed_with_rotation' :
                         riskScore >= 20 ? 'proceed_with_rotation' : 'proceed';

  const holdDays = riskScore >= 70 
    ? getAdaptiveFrivolousHoldDays(ctx.bureau, ctx.pass, 'same_legal_basis_repeat', ctx.settings) 
    : undefined;

  const suggestedStrategy = riskScore >= 40 
    ? getRotatedStrategy(ctx.currentStrategy, ctx.bureau, ctx.pass)
    : undefined;

  return { riskScore, riskLevel, triggerFactors: factors, recommendation, holdDays, suggestedStrategy };
}
```

```typescript
// After grouping glyphs by Y-bucket, before producing lines:
// Check for account number continuation patterns

function stitchAccountNumberContinuations(lines: string[]): string[] {
  const ACCOUNT_PATTERN = /account\s*(?:#|number|num|no\.?)[\s:]*([0-9X\*\-]{4,})/i;
  const CONTINUATION_PATTERN = /^[0-9X\*]{4,20}$/;  // Pure number/mask continuation
  
  const result: string[] = [];
  
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    const accountMatch = line.match(ACCOUNT_PATTERN);
    
    if (accountMatch && i + 1 < lines.length) {
      const nextLine = lines[i + 1].trim();
      
      // If next line looks like an account number continuation
      if (CONTINUATION_PATTERN.test(nextLine)) {
        // Stitch: current line + continuation
        result.push(`${line} ${nextLine}`);
        i++; // skip the continuation line
        continue;
      }
    }
    
    result.push(line);
  }
  
  return result;
}
```

### 5.2 Enhanced Account Number Extraction in `creditParser.ts`

Add a dedicated account number extraction function with multi-pattern matching:

```typescript
// In creditParser.ts — replace single-regex account extraction

const ACCOUNT_NUMBER_PATTERNS = [
  // Bureau-specific formats
  /(?:account\s*(?:#|number|num|no\.?|nbr)[:\s]+)([\dX\*\-\s]{6,25})/i,
  /(?:acct\.?\s*(?:#|no\.?)[:\s]+)([\dX\*\-\s]{6,25})/i,
  
  // Experian: "Account Number: 123456XXXXXXXX"
  /(\d{4,}\s*[Xx\*]{4,}\s*\d{0,4})/,
  
  // TransUnion: suffix only "Account #: 1234"
  /(?:account\s*#:\s*)(\d{4,8})\b/i,
  
  // Equifax: "XXXX-XXXX-XXXX-1234"
  /([Xx\*]{4}[\-\s]?[Xx\*]{4}[\-\s]?[Xx\*]{4}[\-\s]?\d{4})/,
  
  // Generic: any sequence of digits + masks that's plausibly an account number
  /\b(\d{2,4}[Xx\*]{2,}\d{2,4})\b/,
  
  // Last resort: bare digits that look like account numbers (8-20 digits)
  /\b(\d{8,20})\b/,
];

export function extractBestAccountNumber(text: string): string | null {
  for (const pattern of ACCOUNT_NUMBER_PATTERNS) {
    const match = text.match(pattern);
    if (match && match[1]) {
      const candidate = match[1].trim().replace(/\s+/g, '');
      // Must have at least 4 digits or 4 mask characters
      const digitAndMaskCount = (candidate.match(/[\dXx\*]/g) ?? []).length;
      if (digitAndMaskCount >= 4) {
        return candidate;
      }
    }
  }
  return null;
}
```
