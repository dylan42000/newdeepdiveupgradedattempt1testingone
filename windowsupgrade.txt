## 🔴 THE PROBLEM — PLAIN ENGLISH DIAGNOSIS

Your AutoPilot is generating letters where the **sender block (top-left)** — the consumer's name, address, city, state, ZIP — is either blank, showing creditor/bureau data, or showing unfilled placeholder tokens like `{{CONSUMER_NAME}}` or `{{ADDRESS_LINE_1}}`.

**Root Cause Chain:**

```
AppContext.personalInfo ──── EXISTS ✅
     │
     ▼
autopilotEngine.ts ─── runAutopilotCycle() ─── does NOT receive personalInfo ❌
     │
     ▼
letterGeneratorV2.ts ─── generateLetter() ─── personalInfo is undefined/null ❌
     │
     ▼
placeholderService.ts ─── smartFillLetter() ─── fills creditor fields fine ✅
                                             └── fills consumer fields BLANK ❌
     │
     ▼
letterTemplateService.ts ─── buildLetterHTML() ─── SENDER BLOCK EMPTY ❌
```

**The pipeline breaks at the handoff between AppContext and the AutoPilot engine.** The `personalInfo` object never gets injected into the auto-generation chain, so the placeholder service has nothing to fill the consumer fields with — but it CAN fill the creditor/bureau fields because those come from `NegativeItem` objects which ARE correctly passed through. This creates the exact symptom you're seeing: creditor info shows up fine, your own info is missing or wrong.

```typescript
// Builds the multi-line sender address block for the letter header
function buildConsumerFullAddress(p: PersonalInfo): string {
  const lines = [
    p.address,
    p.address2,
    `${p.city}, ${p.state} ${p.zip}`
  ].filter(Boolean);
  return lines.join('\n');
}

// Always mask account numbers in letters — show only last 4
function maskAccountNumber(acct: string | null | undefined): string {
  if (!acct) return '';
  const clean = acct.replace(/\s/g, '');
  if (clean.length <= 4) return clean;
  return `XXXX-XXXX-XXXX-${clean.slice(-4)}`;
}

// Hard-coded bureau dispute mailing addresses
function getBureauAddress(bureau: string): string {
  const addresses: Record<string, string> = {
    'equifax':    'Equifax Information Services LLC\nP.O. Box 740256\nAtlanta, GA 30374-0256',
    'experian':   'Experian\nP.O. Box 4500\nAllen, TX 75013',
    'transunion': 'TransUnion LLC\nConsumer Dispute Center\nP.O. Box 2000\nChester, PA 19016',
  };
  return addresses[bureau.toLowerCase()] || bureau;
}
```

### The fixed `smartFillLetter` function:

```typescript
export function smartFillLetter(
  content: string,
  personalInfo: PersonalInfo | null | undefined,
  items: NegativeItem[],
  bureau: string = '',
  round: number = 1
): { filled: string; unfilledTokens: string[] } {

  // GUARD: If personalInfo is missing, log loudly and use empty defaults
  // DO NOT silently fail — surface the problem
  if (!personalInfo || !personalInfo.firstName) {
    console.error(
      '[placeholderService] CRITICAL: personalInfo is null/undefined or has no firstName. ' +
      'Consumer sender block WILL be empty. AutoPilot must pass personalInfo from AppContext.'
    );
  }

  let filled = content;
  const safeInfo: PersonalInfo = personalInfo ?? {
    firstName: '[CONSUMER FIRST NAME]',
    lastName:  '[CONSUMER LAST NAME]',
    address:   '[CONSUMER ADDRESS]',
    city:      '[CITY]',
    state:     '[STATE]',
    zip:       '[ZIP]',
  };

  // STEP 1: Fill consumer tokens FIRST (highest priority)
  for (const [token, resolver] of Object.entries(CONSUMER_TOKEN_MAP)) {
    const value = resolver(safeInfo);
    filled = filled.replaceAll(token, value);
  }

  // STEP 2: Fill creditor/item tokens from PRIMARY item (index 0)
  const primaryItem = items?.[0];
  if (primaryItem) {
    for (const [token, resolver] of Object.entries(CREDITOR_TOKEN_MAP)) {
      const value = resolver(primaryItem);
      filled = filled.replaceAll(token, value);
    }
  }

  // STEP 3: Fill bureau/system tokens
  const targetBureau = bureau || primaryItem?.creditBureau?.[0] || '';
  for (const [token, resolver] of Object.entries(BUREAU_TOKEN_MAP)) {
    const value = resolver(primaryItem ?? {} as NegativeItem, targetBureau, round);
    filled = filled.replaceAll(token, value);
  }

  // STEP 4: Scan for any remaining unfilled tokens
  const unfilledTokens = scanForUnfilledTokens(filled);
  if (unfilledTokens.length > 0) {
    console.warn('[placeholderService] Unfilled tokens after fill pass:', unfilledTokens);
  }

  return { filled, unfilledTokens };
}

// Scans content for any remaining {{TOKEN}} patterns
export function scanForUnfilledTokens(content: string): string[] {
  const matches = content.match(/\{\{[A-Z_]+\}\}/g) ?? [];
  return [...new Set(matches)];
}
```

```typescript
// This function builds the TOP-LEFT sender address block
// It ONLY uses PersonalInfo — never NegativeItem fields
// If a field is missing, render a visible placeholder, never silently omit

function buildSenderBlock(personalInfo: PersonalInfo): string {
  const name    = `${personalInfo.firstName ?? ''} ${personalInfo.lastName ?? ''}`.trim()
                  || '[YOUR NAME]';
  const addr1   = personalInfo.address   || '[YOUR STREET ADDRESS]';
  const addr2   = personalInfo.address2  || null;
  const cityStateZip = [
    personalInfo.city  || '[CITY]',
    personalInfo.state || '[ST]',
    personalInfo.zip   || '[ZIP]'
  ].join(', ').replace(', ,', ',');

  const phone   = personalInfo.phone ? `<br>Phone: ${personalInfo.phone}` : '';
  const email   = personalInfo.email ? `<br>Email: ${personalInfo.email}` : '';

  return `
    <div class="sender-block" style="
      font-family: 'Times New Roman', Times, serif;
      font-size: 12pt;
      line-height: 1.6;
      margin-bottom: 20px;
    ">
      <strong>${name}</strong><br>
      ${addr1}<br>
      ${addr2 ? addr2 + '<br>' : ''}
      ${cityStateZip}
      ${phone}
      ${email}
    </div>
  `;
}

// This function builds the RECIPIENT block (bureau or furnisher address)
// It ONLY uses bureau/furnisher data — never PersonalInfo
// Clear separation prevents the cross-contamination bug

function buildRecipientBlock(
  bureau: string,
  contactAddress?: string | null
): string {
  const recipientAddress = contactAddress || getBureauAddress(bureau);
  const lines = recipientAddress.split('\n').filter(Boolean);

  return `
    <div class="recipient-block" style="
      font-family: 'Times New Roman', Times, serif;
      font-size: 12pt;
      line-height: 1.6;
      margin-bottom: 20px;
    ">
      ${lines.map(line => `${line}<br>`).join('')}
    </div>
  `;
}
```

## ✅ STEP 8 — THE PREFLIGHT CHECK FUNCTION

Add this utility to `autopilotEngine.ts` that runs before any cycle and validates all prerequisites:

```typescript
interface AutopilotPreflightResult {
  canRun:   boolean;
  errors:   string[];
  warnings: string[];
}

export function runAutopilotPreflight(
  personalInfo: PersonalInfo | null | undefined,
  items: NegativeItem[],
  settings: AutopilotSettings
): AutopilotPreflightResult {

  const errors: string[]   = [];
  const warnings: string[] = [];

  // ── CRITICAL: Personal info completeness ──────────────────────────────
  if (!personalInfo) {
    errors.push('FATAL: personalInfo is null. No letters can be generated. Set up your profile first.');
  } else {
    if (!personalInfo.firstName?.trim())
      errors.push('Missing: First Name (required for sender block)');
    if (!personalInfo.lastName?.trim())
      errors.push('Missing: Last Name (required for sender block)');
    if (!personalInfo.address?.trim())
      errors.push('Missing: Street Address (required for sender block)');
    if (!personalInfo.city?.trim())
      errors.push('Missing: City (required for sender block)');
    if (!personalInfo.state?.trim())
      errors.push('Missing: State (required for sender block)');
    if (!personalInfo.zip?.trim())
      errors.push('Missing: ZIP Code (required for sender block)');

    // Warnings (nice-to-have but not blocking)
    if (!personalInfo.phone?.trim())
      warnings.push('Optional: Phone number not set (recommended for letters)');
    if (!personalInfo.email?.trim())
      warnings.push('Optional: Email not set');
    if (!personalInfo.ssnLast4?.trim())
      warnings.push('Optional: SSN last 4 not set (some bureaus require for verification)');
  }

  // ── Items check ────────────────────────────────────────────────────────
  const activeItems = items.filter(i =>
    i.disputeStatus !== 'Deleted' &&
    i.disputeStatus !== 'Won' &&
    i.disputeStatus !== 'RESOLVED_DELETED'
  );
  if (activeItems.length === 0) {
    errors.push('No active disputable items found. Add items before running AutoPilot.');
  }

  // ── Settings check ─────────────────────────────────────────────────────
  if (!settings.enabled) {
    warnings.push('AutoPilot is disabled in settings. Enable it to run automatically.');
  }

  return {
    canRun:   errors.length === 0,
    errors,
    warnings,
  };
}
```

### Call this in `runAutopilotCycle` at the very top:

```typescript
export async function runAutopilotCycle(params: AutopilotCycleParams): Promise<void> {

  // PREFLIGHT — abort early with clear error if personalInfo is bad
  const preflight = runAutopilotPreflight(
    params.personalInfo,
    params.items,
    params.settings
  );

  if (!preflight.canRun) {
    preflight.errors.forEach(err => {
      params.onLog({
        id: crypto.randomUUID(),
        timestamp: new Date().toISOString(),
        level: 'error',
        message: `[PREFLIGHT FAILED] ${err}`,
        phase: 'pre_flight',
      });
    });
    throw new Error(`AutoPilot preflight failed: ${preflight.errors.join(' | ')}`);
  }

  preflight.warnings.forEach(warn => {
    params.onLog({
      id: crypto.randomUUID(),
      timestamp: new Date().toISOString(),
      level: 'warn',
      message: `[PREFLIGHT WARNING] ${warn}`,
      phase: 'pre_flight',
    });
  });

  // ... rest of cycle logic continues ...
}
```
